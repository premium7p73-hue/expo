// --- شاشة تسجيل الدخول ---
const AuthScreen = () => {
  const [isLogin, setIsLogin] = useState(true);
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [name, setName] = useState('');
  const [loading, setLoading] = useState(false);

  const handleAuth = async () => {
    if(!email || !password) {
      Alert.alert("تنبيه", "الرجاء إدخال البريد وكلمة المرور");
      return;
    }
    setLoading(true);
    try {
      if (isLogin) {
        await signInWithEmailAndPassword(auth, email, password);
      } else {
        const cred = await createUserWithEmailAndPassword(auth, email, password);
        await updateProfile(cred.user, { displayName: name || 'مستخدم جديد' });
      }
    } catch (error) {
      let msg = error.message;
      if (error.code === 'auth/email-already-in-use') msg = 'البريد مسجل مسبقاً';
      if (error.code === 'auth/weak-password') msg = 'كلمة المرور ضعيفة';
      Alert.alert("خطأ", msg);
    } finally {
      setLoading(false);
    }
  };

  return (
    <KeyboardAvoidingView 
      behavior={Platform.OS === "ios" ? "padding" : "height"} 
      style={styles.container}
    >
      <View style={styles.authContainer}>
        <View style={styles.logoContainer}>
          <Ionicons name="mic-circle" size={80} color="#818cf8" />
          <Text style={styles.logoText}>راوي</Text>
          <Text style={styles.subtitle}>صوتك مسموع</Text>
        </View>

        {!isLogin && (
          <TextInput 
            placeholder="الاسم الكامل" 
            placeholderTextColor="#64748b"
            style={styles.input} 
            value={name}
            onChangeText={setName}
            textAlign="right"
          />
        )}
        <TextInput 
          placeholder="البريد الإلكتروني" 
          placeholderTextColor="#64748b"
          style={styles.input} 
          value={email}
          onChangeText={setEmail}
          keyboardType="email-address"
          autoCapitalize="none"
          textAlign="right"
        />
        <TextInput 
          placeholder="كلمة المرور" 
          placeholderTextColor="#64748b"
          style={styles.input} 
          value={password}
          onChangeText={setPassword}
          secureTextEntry
          textAlign="right"
        />

        <TouchableOpacity style={styles.primaryButton} onPress={handleAuth} disabled={loading}>
          {loading ? <ActivityIndicator color="#fff" /> : <Text style={styles.buttonText}>{isLogin ? 'تسجيل الدخول' : 'إنشاء حساب'}</Text>}
        </TouchableOpacity>

        <TouchableOpacity onPress={() => setIsLogin(!isLogin)} style={{marginTop: 20}}>
          <Text style={styles.linkText}>
            {isLogin ? 'ليس لديك حساب؟ انضم إلينا' : 'لديك حساب بالفعل؟ سجل دخولك'}
          </Text>
        </TouchableOpacity>
      </View>
    </KeyboardAvoidingView>
  );
};

// --- بطاقة المنشور ---
const PostCard = ({ item, currentUserId }) => {
  const [liked, setLiked] = useState(false);

  const handleLike = async () => {
    setLiked(!liked);
    try {
      const ref = doc(db, 'posts', item.id);
      await updateDoc(ref, { likes: increment(liked ? -1 : 1) });
    } catch (e) { console.log(e); }
  };

  return (
    <View style={styles.card}>
      <View style={styles.cardHeader}>
        <View style={styles.userInfo}>
          <View style={styles.avatar}>
            <Text style={styles.avatarText}>{item.username?.[0]?.toUpperCase() || '?'}</Text>
          </View>
          <View>
            <Text style={styles.username}>{item.username}</Text>
            <Text style={styles.date}>الآن</Text>
          </View>
        </View>
        <View style={styles.typeBadge}>
          <Ionicons 
            name={item.type === 'video' ? 'videocam' : item.type === 'audio' ? 'mic' : 'text'} 
            size={12} 
            color="#94a3b8" 
          />
        </View>
      </View>

      <Text style={styles.postContent}>{item.content}</Text>

      {/* منطقة الوسائط (Placeholder) */}
      {(item.type === 'video' || item.type === 'audio') && (
        <View style={styles.mediaPlaceholder}>
          <Ionicons 
            name={item.type === 'video' ? 'play-circle' : 'musical-notes'} 
            size={50} 
            color="#fff" 
            style={{opacity: 0.8}}
          />
          <Text style={{color: '#64748b', marginTop: 10, fontSize: 12}}>
            {item.type === 'video' ? 'فيديو مرفق' : 'مقطع صوتي'}
          </Text>
        </View>
      )}

      <View style={styles.cardFooter}>
        <TouchableOpacity style={styles.actionBtn} onPress={handleLike}>
          <Ionicons name={liked ? "heart" : "heart-outline"} size={22} color={liked ? "#ec4899" : "#94a3b8"} />
          <Text style={[styles.actionText, liked && {color: '#ec4899'}]}>{item.likes || 0}</Text>
        </TouchableOpacity>
        
        <TouchableOpacity style={styles.actionBtn}>
          <Ionicons name="chatbubble-outline" size={20} color="#94a3b8" />
          <Text style={styles.actionText}>تعليق</Text>
        </TouchableOpacity>
      </View>
    </View>
  );
};

// --- التطبيق الرئيسي ---
export default function App() {
  const [user, setUser] = useState(null);
  const [posts, setPosts] = useState([]);
  const [modalVisible, setModalVisible] = useState(false);
  const [newPostContent, setNewPostContent] = useState('');
  const [postType, setPostType] = useState('text'); // text, audio, video

  // مراقبة الدخول
  useEffect(() => {
    if (!auth) return;
    const unsubscribe = onAuthStateChanged(auth, setUser);
    return unsubscribe;
  }, []);

  // جلب المنشورات
  useEffect(() => {
    if (!user || !db) return;
    const q = query(collection(db, 'posts'), orderBy('createdAt', 'desc'));
    const unsubscribe = onSnapshot(q, (snapshot) => {
      setPosts(snapshot.docs.map(d => ({ id: d.id, ...d.data() })));
    });
    return unsubscribe;
  }, [user]);

  const handlePublish = async () => {
    if (!newPostContent.trim()) return;
    try {
      await addDoc(collection(db, 'posts'), {
        uid: user.uid,
        username: user.displayName || 'مجهول',
        content: newPostContent,
        type: postType,
        likes: 0,
        createdAt: serverTimestamp()
      });
      setNewPostContent('');
      setModalVisible(false);
    } catch (e) {
      Alert.alert("خطأ", "فشل النشر: " + e.message);
    }
  };

  if (!user) return <AuthScreen />;

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar style="light" />
      
      {/* الهيدر */}
      <View style={styles.header}>
        <Text style={styles.headerTitle}>راوي</Text>
        <TouchableOpacity onPress={() => signOut(auth)}>
          <Ionicons name="log-out-outline" size={24} color="#94a3b8" />
        </TouchableOpacity>
      </View>

      {/* قائمة المنشورات */}
      <FlatList
        data={posts}
        keyExtractor={item => item.id}
        renderItem={({ item }) => <PostCard item={item} currentUserId={user.uid} />}
        contentContainerStyle={{ padding: 16, paddingBottom: 100 }}
        ListEmptyComponent={
          <View style={{alignItems: 'center', marginTop: 100}}>
            <Ionicons name="library-outline" size={60} color="#334155" />
            <Text style={{color: '#64748b', marginTop: 10}}>لا توجد منشورات بعد</Text>
          </View>
        }
      />

      {/* زر إضافة منشور */}
      <TouchableOpacity style={styles.fab} onPress={() => setModalVisible(true)}>
        <Ionicons name="add" size={32} color="#fff" />
      </TouchableOpacity>

      {/* نافذة إنشاء منشور */}
      <Modal visible={modalVisible} animationType="slide" transparent={true} onRequestClose={() => setModalVisible(false)}>
        <View style={styles.modalOverlay}>
          <KeyboardAvoidingView behavior={Platform.OS === "ios" ? "padding" : "height"} style={styles.modalContent}>
            
            <View style={styles.modalHeader}>
              <Text style={styles.modalTitle}>منشور جديد</Text>
              <TouchableOpacity onPress={() => setModalVisible(false)}>
                <Ionicons name="close" size={24} color="#94a3b8" />
              </TouchableOpacity>
            </View>

            <TextInput 
              style={styles.postInput}
              placeholder="بماذا تفكر؟"
              placeholderTextColor="#64748b"
              multiline
              value={newPostContent}
              onChangeText={setNewPostContent}
              textAlign="right"
            />

            {/* أزرار اختيار النوع */}
            <View style={styles.typeSelector}>
              <TouchableOpacity onPress={() => setPostType('text')} style={[styles.typeBtn, postType === 'text' && styles.activeTypeBtn]}>
                <Ionicons name="text" size={20} color={postType === 'text' ? '#fff' : '#94a3b8'} />
              </TouchableOpacity>
              <TouchableOpacity onPress={() => setPostType('audio')} style={[styles.typeBtn, postType === 'audio' && styles.activeTypeBtn]}>
                <Ionicons name="mic" size={20} color={postType === 'audio' ? '#fff' : '#94a3b8'} />
              </TouchableOpacity>
              <TouchableOpacity onPress={() => setPostType('video')} style={[styles.typeBtn, postType === 'video' && styles.activeTypeBtn]}>
                <Ionicons name="videocam" size={20} color={postType === 'video' ? '#fff' : '#94a3b8'} />
              </TouchableOpacity>
            </View>

            <TouchableOpacity style={styles.publishBtn} onPress={handlePublish}>
              <Text style={styles.publishBtnText}>نشر الآن</Text>
              <Ionicons name="send" size={18} color="#fff" style={{marginLeft: 8}} />
            </TouchableOpacity>

          </KeyboardAvoidingView>
        </View>
      </Modal>

    </SafeAreaView>
  );
}
